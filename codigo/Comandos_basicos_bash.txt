Como vemos los permisos 
// ls -l archivos.txt # me da información de permisos 
Anatomía de los permisos 
# del video [-rw-r--r--]
[-] # guion archivo normal [l] # archivo simbólico [b] # bloque [s] # dispositivo de carácter etc
[rw-] # permisos de usuario, tiene permiso de lectura, escritura pero no tiene permiso de ejecución [rwx] todos los permisos 
[r--] # permisos del grupo 
[r--] # permisos de otros 
si aparece [-] no tiene ese permiso otorgado 
#modo octal
r = 4
w = 2
x = 1
## modificar permisos 
// chmod # modificar el permiso 
// chmod u+ añadir permiso u- quitar permiso  
// chmod u+ archivo.txt # le di permiso al usuario [u] de ejecución 
# modo octal
// chmod 777 archivo.txt # primer [7] hace referencia al grupo de usuario-propietario el segundo [7] usuarios y el tercer [7] a otros 
// chmod 644 archivo.txt # cada numero es hace referencia a un permiso 
## cambiar el propietario del grupo 
// chown dario:grupo archivo.txt # cambiar el propietario y de grupo
